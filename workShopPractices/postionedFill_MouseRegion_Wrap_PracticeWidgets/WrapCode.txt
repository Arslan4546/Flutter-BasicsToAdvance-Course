import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        backgroundColor: Colors.grey[200],
        appBar: AppBar(
          title: Text("Beautiful Wrap Design"),
          backgroundColor: Colors.deepPurple,
          centerTitle: true,
        ),
        body: Padding(
          padding: const EdgeInsets.all(12.0),
          child: WrapWidgetExample(),
        ),
      ),
    );
  }
}

class WrapWidgetExample extends StatefulWidget {
  @override
  _WrapWidgetExampleState createState() => _WrapWidgetExampleState();
}

class _WrapWidgetExampleState extends State<WrapWidgetExample> {
  List<Color> cardColors = [
    Colors.blueAccent,
    Colors.orangeAccent,
    Colors.greenAccent,
    Colors.purpleAccent,
    Colors.redAccent,
    Colors.teal,
    Colors.amber,
    Colors.pinkAccent,
  ];

  @override
  Widget build(BuildContext context) {
    return Wrap(
      spacing: 10, // Space between each card
      runSpacing: 10, // Space between rows
      alignment: WrapAlignment.center,
      children: List.generate(cardColors.length, (index) {
        return GestureDetector(
          onTap: () {
            setState(() {
              cardColors[index] = cardColors[index].withOpacity(
                  (cardColors[index].opacity == 1.0) ? 0.5 : 1.0);
            });
          },
          child: AnimatedContainer(
            duration: Duration(milliseconds: 300),
            width: 100,
            height: 100,
            decoration: BoxDecoration(
              color: cardColors[index],
              borderRadius: BorderRadius.circular(15),
              boxShadow: [
                BoxShadow(
                  color: Colors.black26,
                  blurRadius: 10,
                  offset: Offset(3, 5),
                ),
              ],
            ),
            child: Center(
              child: Text(
                "Card ${index + 1}",
                style: TextStyle(
                  color: Colors.white,
                  fontWeight: FontWeight.bold,
                ),
              ),
            ),
          ),
        );
      }),
    );
  }
}
