import 'package:flutter/material.dart';

void main() => runApp(SolarLunarEclipseApp());

class SolarLunarEclipseApp extends StatefulWidget {
  @override
  _SolarLunarEclipseAppState createState() => _SolarLunarEclipseAppState();
}

class _SolarLunarEclipseAppState extends State<SolarLunarEclipseApp> {
  bool isSolarEclipse = true;

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        backgroundColor: Colors.black,
        appBar: AppBar(
          title: Text(isSolarEclipse ? 'Solar Eclipse' : 'Lunar Eclipse'),
          backgroundColor: Colors.deepPurple,
          actions: [
            IconButton(
              icon: Icon(Icons.swap_horiz),
              onPressed: () {
                setState(() {
                  isSolarEclipse = !isSolarEclipse;
                });
              },
            ),
          ],
        ),
        body: Center(
          child: isSolarEclipse ? SolarEclipse() : LunarEclipse(),
        ),
      ),
    );
  }
}

class SolarEclipse extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Stack(
      alignment: Alignment.center,
      children: [
        // Sun container
        Container(
          width: 150,
          height: 150,
          decoration: BoxDecoration(
            color: Colors.yellow,
            shape: BoxShape.circle,
            boxShadow: [
              BoxShadow(
                color: Colors.orangeAccent,
                blurRadius: 50,
                spreadRadius: 20,
              ),
            ],
          ),
        ),
        // Moon container
        Positioned(
          left: 20, // Adjust the value to control the eclipse overlap
          child: Container(
            width: 140,
            height: 140,
            decoration: BoxDecoration(
              color: Colors.black,
              shape: BoxShape.circle,
              boxShadow: [
                BoxShadow(
                  color: Colors.black45,
                  blurRadius: 15,
                  spreadRadius: 5,
                ),
              ],
            ),
          ),
        ),
      ],
    );
  }
}

class LunarEclipse extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Stack(
      alignment: Alignment.center,
      children: [
        // Moon container
        Container(
          width: 150,
          height: 150,
          decoration: BoxDecoration(
            color: Colors.grey[300],
            shape: BoxShape.circle,
            boxShadow: [
              BoxShadow(
                color: Colors.grey,
                blurRadius: 30,
                spreadRadius: 10,
              ),
            ],
          ),
        ),
        // Earth's shadow (simulate with black container)
        Positioned(
          left: 20, // Adjust the value to control the shadow overlap
          child: Container(
            width: 160,
            height: 160,
            decoration: BoxDecoration(
              color: Colors.black,
              shape: BoxShape.circle,
              boxShadow: [
                BoxShadow(
                  color: Colors.black45,
                  blurRadius: 20,
                  spreadRadius: 5,
                ),
              ],
            ),
          ),
        ),
      ],
    );
  }
}
