import 'package:flutter/material.dart';

void main() {
  runApp(const MyApp());
}

class MyApp extends StatelessWidget {
  const MyApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'Smooth Circular Expansion',
      debugShowCheckedModeBanner: false,
      theme: ThemeData(
        colorScheme: ColorScheme.fromSeed(seedColor: Colors.blue),
        useMaterial3: true,
      ),
      home: const FirstScreen(),
    );
  }
}

class CircleRectTween extends RectTween {
  CircleRectTween({required Rect begin, required Rect end})
      : super(begin: begin, end: end);

  @override
  Rect lerp(double t) {
    final adjustedT = Curves.easeOut.transform(t);

    final beginCenter = begin!.center;

    final maxRadius =
        end!.width > end!.height ? end!.width * 1.5 : end!.height * 1.5;

    final buttonRadius = begin!.width / 2;

    final radius = buttonRadius + (maxRadius - buttonRadius) * adjustedT;

    return Rect.fromCircle(center: beginCenter, radius: radius);
  }
}

class FirstScreen extends StatelessWidget {
  const FirstScreen({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.white,
      appBar: AppBar(

        child: Column(

          children: [
            const Text(
              'Tap the black button below',
              style: TextStyle(fontSize: 18, fontWeight: FontWeight.w500),
            ),
            const SizedBox(height: 30),
            GestureDetector(
              onTap: () {
                Navigator.of(context).push(
                  PageRouteBuilder(
                    transitionDuration: const Duration(milliseconds: 1500),
                    pageBuilder: (context, animation, secondaryAnimation) {
                      return FadeTransition(
                        opacity: animation,
                        child: const SecondScreen(),
                      );
                    },
                  ),
                );
              },
              child: Hero(
                tag: 'circleTransition',
                createRectTween: (begin, end) {
                  return CircleRectTween(begin: begin!, end: end!);
                },
                child: Container(
                  width: 80,
                  height: 80,
                  decoration: BoxDecoration(
                    color: Colors.black,
                    shape: BoxShape.circle,
                    boxShadow: [
                      BoxShadow(
                        color: Colors.black.withOpacity(0.3),
                        blurRadius: 10,
                        spreadRadius: 2,
                      ),
                    ],
                  ),
                  child: const Icon(
                    Icons.touch_app,
                    color: Colors.white,
                    size: 36,
                  ),
                ),
              ),
            ),
            const SizedBox(height: 50),
            Container(
              padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 15),
              decoration: BoxDecoration(
                color: Colors.blue.shade50,
                borderRadius: BorderRadius.circular(12),

              ),

                'Watch the button slowly expand to cover the screen',
                style: TextStyle(fontSize: 16, color: Colors.blue),

          ],

      ),
    );
  }
}
