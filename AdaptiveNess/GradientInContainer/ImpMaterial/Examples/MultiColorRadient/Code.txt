import 'package:flutter/material.dart';

void main() {
  runApp(MyApp());
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: Scaffold(
        appBar: AppBar(title: Text("Gradient Designs")),
        body: Center(
         child: MultiColorLinearGradientExample(),
        ),
      ),
    );
  }
}

class MultiColorLinearGradientExample extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Container(
      width: 300,
      height: 300,
      decoration: BoxDecoration(
        gradient: LinearGradient(
          colors: [Colors.red, Colors.pink, Colors.purple, Colors.blue],
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          stops: [0.0, 0.3, 0.7, 1.0],
        ),
        borderRadius: BorderRadius.circular(15),
      ),
      child: Center(
        child: Text(
          'Multi-Color Linear Gradient',
          style: TextStyle(color: Colors.white, fontSize: 20),
        ),
      ),
    );
  }
}
